#### Server 端提示词拼接

```javascript
// server/services/promptService.js

class PromptService {
  /**
   * 为 Claude SDK 调用构建完整提示词
   */
  async buildPromptForSDK(novelId, chapterId, userMessage) {
    const novel = await this.getNovel(novelId);
    const chapter = await this.getChapter(novelId, chapterId);
    const stateFiles = await this.getStateFiles(novelId);

    // 系统提示词层（固定）
    const systemPrompt = this.getSystemPrompt();

    // 运行时上下文层（动态）
    const runtimeContext = this.buildRuntimeContext({
      novel,
      chapter,
      stateFiles,
      cursorState: await this.getCursorState(novelId)
    });

    return {
      systemPrompt: systemPrompt + '\n\n' + runtimeContext,
      userMessage: userMessage,
      // 额外的 SDK 选项
      options: {
        cwd: novel.projectPath,
        systemPrompt: {
          type: 'preset',
          preset: 'claude_code'  // 使用 SDK 的 preset 模式
        },
        settingSources: ['project', 'user', 'local']
      }
    };
  }

  buildRuntimeContext(context) {
    const { novel, chapter, stateFiles, cursorState } = context;

    return `
━━━━━━━━━━━━━━━━━━━━━━━━
【当前项目状态】
━━━━━━━━━━━━━━━━━━━━━━━━

项目名称: ${novel.name}
当前章节: 第 ${chapter.number} 章 - ${chapter.title}
写作阶段: ${cursorState.phase || 'idle'}
已完成章节: ${novel.completedChapterCount || 0}/${novel.totalChapterCount || '?'}

━━━━━━━━━━━━━━━━━━━━━━━━
【可用状态文件】
━━━━━━━━━━━━━━━━━━━━━━━━

${stateFiles.map(f => `- ${f.name}: ${f.description}`).join('\n')}

━━━━━━━━━━━━━━━━━━━━━━━━
【当前章节信息】
━━━━━━━━━━━━━━━━━━━━━━━━

章节标题: ${chapter.title}
预计字数: ${chapter.targetWordCount || '未设定'}
当前字数: ${chapter.currentWordCount || 0}
章节状态: ${chapter.status} (待写/规划中/写作中/已完成)

${chapter.outline ? `
章节大纲:
${chapter.outline}
` : '⚠️ 章节大纲缺失，请先创建大纲'}
    `;
  }
}
```
